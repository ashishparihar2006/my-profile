  let state = initialState;
  const listeners = new Set();
  return {
    getState: () => state,
    subscribe: (fn) => (listeners.add(fn), () => listeners.delete(fn)),
    setState: (update) => {

      listeners.forEach((cb) => cb(state));

  };
